Get-ChildItem -Recurse -Filter *.import | ForEach-Object {
    $importFile = $_.FullName
    $original = $importFile -replace '\.import$', ''

    if (-not (Test-Path $original)) {
        Write-Host "Deleting orphan import: $importFile"
        Remove-Item $importFile
    }
}
